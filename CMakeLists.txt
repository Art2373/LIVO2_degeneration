cmake_minimum_required(VERSION 3.10)
project(livo2_degeneration)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -pthread -fexceptions")
# Preserve IEEE finite-value checks. Do not use -ffast-math/-Ofast here.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|aarch64|ARM|AARCH64)")
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -mtune=native")
  add_definitions(-DARM_ARCH)
else()
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -mtune=native -funroll-loops")
  add_definitions(-DX86_ARCH)
endif()
include(ProcessorCount)
ProcessorCount(N)
if(N GREATER 4)
  set(PROC_NUM 4)
elseif(N GREATER 1)
  set(PROC_NUM ${N})
else()
  set(PROC_NUM 1)
endif()
if(PROC_NUM GREATER 1)
  add_definitions(-DMP_EN -DMP_PROC_NUM=${PROC_NUM})
else()
  add_definitions(-DMP_PROC_NUM=1)
endif()
find_package(OpenMP QUIET)
find_package(mimalloc QUIET)
if(OpenMP_CXX_FOUND)
  add_compile_options(${OpenMP_CXX_FLAGS})
endif()

# This switch changes the recorded Livox wire type, not a topic name.
option(FASTLIVO_USE_LIVOX_ROS_DRIVER2 "Use Livox ROS Driver2 CustomMsg (MID360)" ON)
set(LIVOX_DEPENDENCY)
if(FASTLIVO_USE_LIVOX_ROS_DRIVER2)
  list(APPEND LIVOX_DEPENDENCY livox_ros_driver2)
  add_definitions(-DFASTLIVO_USE_LIVOX_ROS_DRIVER2)
endif()
find_package(catkin REQUIRED COMPONENTS geometry_msgs nav_msgs sensor_msgs roscpp rospy
  std_msgs pcl_ros tf message_generation eigen_conversions vikit_common vikit_ros
  cv_bridge image_transport message_filters ${LIVOX_DEPENDENCY})
catkin_python_setup()
add_message_files(FILES FusionModel.msg)
generate_messages(DEPENDENCIES std_msgs geometry_msgs sensor_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Sophus REQUIRED)
find_package(Boost REQUIRED COMPONENTS thread filesystem system)
catkin_package(INCLUDE_DIRS include
  LIBRARIES degeneration_laser_mapping degeneration_vio degeneration_lio degeneration_pre degeneration_imu_proc
  CATKIN_DEPENDS geometry_msgs nav_msgs sensor_msgs roscpp rospy std_msgs pcl_ros tf
    message_runtime eigen_conversions cv_bridge vikit_common vikit_ros image_transport message_filters ${LIVOX_DEPENDENCY}
  DEPENDS EIGEN3 PCL OpenCV Sophus)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS}
  ${OpenCV_INCLUDE_DIRS} ${Sophus_INCLUDE_DIRS})
add_library(degeneration_vio src/vio.cpp src/frame.cpp src/visual_point.cpp)
add_library(degeneration_lio src/voxel_map.cpp)
add_library(degeneration_pre src/preprocess.cpp)
add_library(degeneration_imu_proc src/IMU_Processing.cpp)
add_library(degeneration_laser_mapping src/LIVMapper.cpp src/measurement_scheduler.cpp src/fusion_model.cpp)
set(DEGEN_LIBS degeneration_vio degeneration_lio degeneration_pre degeneration_imu_proc degeneration_laser_mapping)
foreach(target ${DEGEN_LIBS})
  add_dependencies(${target} ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
  target_link_libraries(${target} ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBRARIES} ${Sophus_LIBRARIES} ${Boost_LIBRARIES})
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${target} OpenMP::OpenMP_CXX)
  endif()
endforeach()
# VIO calls VoxelOctoTree::find_correspond; carry its real library dependency.
target_link_libraries(degeneration_vio degeneration_lio)
add_executable(fusion_mapping src/main.cpp)
add_dependencies(fusion_mapping ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(fusion_mapping degeneration_laser_mapping degeneration_vio degeneration_lio degeneration_pre degeneration_imu_proc
  ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBRARIES} ${Sophus_LIBRARIES} ${Boost_LIBRARIES})
if(OpenMP_CXX_FOUND)
  target_link_libraries(fusion_mapping OpenMP::OpenMP_CXX)
endif()
if(mimalloc_FOUND)
  target_link_libraries(fusion_mapping mimalloc)
endif()
catkin_install_python(PROGRAMS scripts/publish_camera_info.py scripts/fusion_parameter_provider.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS fusion_mapping ${DEGEN_LIBS}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

# Project tests are independent of vendored packages' optional test suites.
option(LIVO2_BUILD_TESTS "Build frontend semantic/boundary regression tests" ON)
if(LIVO2_BUILD_TESTS)
  enable_testing()
  foreach(test_name test_visual_update_result test_visual_lifecycle test_image_retention test_sensor_time test_vio_feedback_integration test_wall_clock_schedule)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_compile_options(${test_name} PRIVATE -UNDEBUG)
    add_dependencies(${test_name} ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
    if(test_name STREQUAL "test_vio_feedback_integration")
      target_link_libraries(${test_name} degeneration_vio ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBRARIES} ${Sophus_LIBRARIES} ${Boost_LIBRARIES})
    else()
      target_link_libraries(${test_name} ${catkin_LIBRARIES} ${Boost_LIBRARIES})
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
endif()
